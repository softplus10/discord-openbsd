cmake_minimum_required(VERSION 3.24)
project(DiscordOpenBSD LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets WebEngineWidgets)
set(APP_VERSION "dev" CACHE STRING "Release version shared by wrapper and updater")
add_executable(discord-openbsd src/main.cpp src/navigation.h)
target_link_libraries(discord-openbsd PRIVATE Qt6::Widgets Qt6::WebEngineWidgets)
target_compile_definitions(discord-openbsd PRIVATE APP_VERSION="${APP_VERSION}")
if(CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
  target_link_options(discord-openbsd PRIVATE -Wl,-z,wxneeded -Wl,-z,nobtcfi)
endif()
add_executable(engine-version src/engine-version.cpp)
target_link_libraries(engine-version PRIVATE Qt6::WebEngineWidgets)
enable_testing()
add_executable(navigation-test tests/navigation.cpp)
target_include_directories(navigation-test PRIVATE src)
target_link_libraries(navigation-test PRIVATE Qt6::Core)
add_test(NAME navigation COMMAND navigation-test)
install(TARGETS discord-openbsd RUNTIME DESTINATION bin)
